```python
import cv2
import numpy as np
import os
import time
from datetime import datetime

# ==========================================================
# SIMPLE FIRE AND SMOKE DETECTION SYSTEM
# OpenCV only - NO AI / NO DATASET
# ==========================================================

# Create alert folder
os.makedirs("alerts", exist_ok=True)

# ----------------------------------------------------------
# SETTINGS
# ----------------------------------------------------------

FIRE_THRESHOLD = 1.0
SMOKE_THRESHOLD = 8.0

ALERT_DELAY = 5

last_alert_time = 0


# ----------------------------------------------------------
# FIRE DETECTION
# ----------------------------------------------------------

def detect_fire(frame):

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Red / Orange / Yellow fire
    lower_fire = np.array([0, 100, 150])
    upper_fire = np.array([40, 255, 255])

    fire_mask = cv2.inRange(
        hsv,
        lower_fire,
        upper_fire
    )

    # Remove noise
    kernel = np.ones((5, 5), np.uint8)

    fire_mask = cv2.morphologyEx(
        fire_mask,
        cv2.MORPH_OPEN,
        kernel
    )

    fire_mask = cv2.morphologyEx(
        fire_mask,
        cv2.MORPH_CLOSE,
        kernel
    )

    fire_pixels = cv2.countNonZero(fire_mask)

    total_pixels = frame.shape[0] * frame.shape[1]

    fire_percentage = (
        fire_pixels / total_pixels
    ) * 100

    return fire_percentage, fire_mask


# ----------------------------------------------------------
# SMOKE DETECTION
# ----------------------------------------------------------

def detect_smoke(frame):

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Gray / white smoke
    lower_smoke = np.array([0, 0, 60])
    upper_smoke = np.array([180, 80, 210])

    smoke_mask = cv2.inRange(
        hsv,
        lower_smoke,
        upper_smoke
    )

    # Remove noise
    kernel = np.ones((7, 7), np.uint8)

    smoke_mask = cv2.morphologyEx(
        smoke_mask,
        cv2.MORPH_OPEN,
        kernel
    )

    smoke_mask = cv2.morphologyEx(
        smoke_mask,
        cv2.MORPH_CLOSE,
        kernel
    )

    smoke_pixels = cv2.countNonZero(smoke_mask)

    total_pixels = frame.shape[0] * frame.shape[1]

    smoke_percentage = (
        smoke_pixels / total_pixels
    ) * 100

    return smoke_percentage, smoke_mask


# ----------------------------------------------------------
# WEBCAM
# ----------------------------------------------------------

cap = cv2.VideoCapture(0)

if not cap.isOpened():

    print("ERROR: Camera could not be opened.")

    exit()


print("=" * 50)
print(" FIRE AND SMOKE DETECTION SYSTEM")
print("=" * 50)
print("Camera started.")
print("Press Q to exit.")
print("=" * 50)


# ----------------------------------------------------------
# MAIN LOOP
# ----------------------------------------------------------

while True:

    ret, frame = cap.read()

    if not ret:

        print("ERROR: Could not read camera.")
        break


    # Resize
    frame = cv2.resize(
        frame,
        (960, 540)
    )


    # ------------------------------------------------------
    # FIRE
    # ------------------------------------------------------

    fire_percentage, fire_mask = detect_fire(frame)


    # ------------------------------------------------------
    # SMOKE
    # ------------------------------------------------------

    smoke_percentage, smoke_mask = detect_smoke(frame)


    # ------------------------------------------------------
    # DECISION
    # ------------------------------------------------------

    status = "NORMAL"

    if fire_percentage >= FIRE_THRESHOLD:

        status = "FIRE DETECTED"

    elif smoke_percentage >= SMOKE_THRESHOLD:

        status = "SMOKE DETECTED"


    # ------------------------------------------------------
    # DISPLAY STATUS
    # ------------------------------------------------------

    if status == "NORMAL":

        cv2.putText(
            frame,
            "STATUS: NORMAL",
            (30, 50),
            cv2.FONT_HERSHEY_SIMPLEX,
            1,
            (0, 255, 0),
            3
        )

    else:

        cv2.putText(
            frame,
            "WARNING: " + status,
            (30, 50),
            cv2.FONT_HERSHEY_SIMPLEX,
            1,
            (0, 0, 255),
            3
        )


        # --------------------------------------------------
        # SAVE ALERT
        # --------------------------------------------------

        current_time = time.time()

        if current_time - last_alert_time >= ALERT_DELAY:

            timestamp = datetime.now().strftime(
                "%Y%m%d_%H%M%S"
            )

            filename = (
                "alerts/alert_"
                + timestamp
                + ".jpg"
            )

            cv2.imwrite(
                filename,
                frame
            )

            print()
            print("======================================")
            print("🚨 ALERT:", status)
            print(
                "Fire:",
                round(fire_percentage, 2),
                "%"
            )
            print(
                "Smoke:",
                round(smoke_percentage, 2),
                "%"
            )
            print("Saved:", filename)
            print("======================================")

            last_alert_time = current_time


    # ------------------------------------------------------
    # DISPLAY VALUES
    # ------------------------------------------------------

    cv2.putText(
        frame,
        f"Fire Level: {fire_percentage:.2f}%",
        (30, 100),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7,
        (255, 255, 255),
        2
    )


    cv2.putText(
        frame,
        f"Smoke Level: {smoke_percentage:.2f}%",
        (30, 135),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7,
        (255, 255, 255),
        2
    )


    # ------------------------------------------------------
    # SHOW CAMERA
    # ------------------------------------------------------

    cv2.imshow(
        "Fire & Smoke Detection System",
        frame
    )


    # ------------------------------------------------------
    # QUIT
    # ------------------------------------------------------

    key = cv2.waitKey(1) & 0xFF

    if key == ord("q"):

        break


# ----------------------------------------------------------
# CLOSE
# ----------------------------------------------------------

cap.release()

cv2.destroyAllWindows()

print("System stopped.")
```
